cmake_minimum_required(VERSION 3.13)
option(HELION_HOST "Build the SDL2 desktop player" OFF)
option(HELION_DMA "DMA background transfer" ON)
option(HELION_INTERP "Core-0 SIO texture addressing and palette blending" ON)
# Measurement only, never a release build: audio_pump() keeps the ring, the
# DMA polling and the chunking but renders silence instead of calling the
# synth, which prices the pump machinery on its own. (Overscan)
option(HELION_SILENT_PUMP "Run audio_pump() with no DSP, for timing only" OFF)
add_compile_definitions(HELION_DMA=$<BOOL:${HELION_DMA}> HELION_INTERP=$<BOOL:${HELION_INTERP}> HELION_SILENT_PUMP=$<BOOL:${HELION_SILENT_PUMP}>)
if(HELION_HOST)
    project(helion C)
    set(CMAKE_C_STANDARD 11)
    find_package(SDL2 CONFIG REQUIRED)
    add_executable(helion host/main.c render.c assets.c synth.c song.c)
    target_include_directories(helion PRIVATE .)
    target_compile_options(helion PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(helion PRIVATE SDL2::SDL2 m)
    add_executable(helion_check tools/check.c render.c assets.c synth.c song.c)
    target_include_directories(helion_check PRIVATE .)
    target_compile_options(helion_check PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation -ftrapv)
    target_link_libraries(helion_check PRIVATE m)
    enable_testing()
    add_test(NAME whole_production COMMAND helion_check)
else()
    set(PICO_BOARD pico2 CACHE STRING "RP2350 board")
    set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Cortex-M33 with FPU")
    if(NOT PICO_SDK_PATH)
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT PICO_EXTRAS_PATH)
        set(PICO_EXTRAS_PATH "$ENV{PICO_EXTRAS_PATH}")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
    include(${PICO_EXTRAS_PATH}/external/pico_extras_import.cmake)
    project(helion C CXX ASM)
    set(CMAKE_C_STANDARD 11)
    pico_sdk_init()
    if(NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "HELION requires RP2350 Cortex-M33 (rp2350-arm-s)")
    endif()
    add_executable(helion main.c video.c audio_pwm.c render.c assets.c synth.c song.c)
    target_include_directories(helion PRIVATE .)
    target_compile_definitions(helion PRIVATE PICO_BUILD=1
        PICO_SCANVIDEO_MAX_SCANLINE_BUFFER_WORDS=324
        PICO_SCANVIDEO_SCANLINE_BUFFER_COUNT=12
        PICO_STACK_SIZE=0x1000 PICO_CORE1_STACK_SIZE=0x1000
        # A panic that keeps USB alive (main.c): the stock one ends in a
        # breakpoint, which without a debugger becomes a HardFault with every
        # interrupt masked, and the board then needs a hand on the cable.
        PICO_PANIC_FUNCTION=helion_panic)
    target_compile_options(helion PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(helion pico_stdlib pico_multicore pico_scanvideo_dpi hardware_pwm hardware_dma m)
    if(HELION_INTERP)
        target_link_libraries(helion hardware_interp)
    endif()
    pico_enable_stdio_usb(helion 1)
    # UART defaults collide with VGA. USB CDC is the sole telemetry channel.
    pico_enable_stdio_uart(helion 0)
    pico_add_extra_outputs(helion)
endif()
