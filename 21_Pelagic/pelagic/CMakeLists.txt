cmake_minimum_required(VERSION 3.13)
option(PELAGIC_HOST "Build the SDL2 desktop player" OFF)
option(PELAGIC_SMOOTH "Optional bilinear environment and premultiplied-alpha ray filtering" OFF)
option(PELAGIC_INTERP "Core-0 INTERP0 UV stepping (host uses a software model)" OFF)
add_compile_definitions(PELAGIC_SMOOTH=$<BOOL:${PELAGIC_SMOOTH}> PELAGIC_INTERP=$<BOOL:${PELAGIC_INTERP}>)
if(PELAGIC_HOST)
    project(pelagic C)
    set(CMAKE_C_STANDARD 11)
    find_package(SDL2 CONFIG REQUIRED)
    add_executable(pelagic host/main.c render.c assets.c synth.c song.c)
    target_include_directories(pelagic PRIVATE .)
    target_compile_options(pelagic PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(pelagic PRIVATE SDL2::SDL2 m)
    add_executable(pelagic_check tools/check.c render.c assets.c synth.c song.c)
    target_include_directories(pelagic_check PRIVATE .)
    target_compile_options(pelagic_check PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation -ftrapv)
    target_link_libraries(pelagic_check PRIVATE m)
    enable_testing()
    add_test(NAME whole_production COMMAND pelagic_check)
    add_executable(pelagic_sampling_check tools/sampling_check.c)
    target_include_directories(pelagic_sampling_check PRIVATE .)
    target_compile_options(pelagic_sampling_check PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation -ftrapv)
    add_test(NAME texture_sampling COMMAND pelagic_sampling_check)
else()
    set(PICO_BOARD pico2 CACHE STRING "RP2350 board")
    set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Cortex-M33 with FPU")
    if(NOT PICO_SDK_PATH)
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT PICO_EXTRAS_PATH)
        set(PICO_EXTRAS_PATH "$ENV{PICO_EXTRAS_PATH}")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
    include(${PICO_EXTRAS_PATH}/external/pico_extras_import.cmake)
    project(pelagic C CXX ASM)
    set(CMAKE_C_STANDARD 11)
    pico_sdk_init()
    if(NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "PELAGIC requires RP2350 Cortex-M33 (rp2350-arm-s)")
    endif()
    add_executable(pelagic main.c video.c audio_pwm.c render.c assets.c synth.c song.c)
    target_include_directories(pelagic PRIVATE .)
    target_compile_definitions(pelagic PRIVATE PICO_BUILD=1
        PICO_SCANVIDEO_MAX_SCANLINE_BUFFER_WORDS=324
        PICO_SCANVIDEO_SCANLINE_BUFFER_COUNT=12
        PICO_STACK_SIZE=0x1000 PICO_CORE1_STACK_SIZE=0x1000
        # A panic that keeps USB alive (main.c): the stock one ends in a
        # breakpoint, which without a debugger becomes a HardFault with every
        # interrupt masked, and the board then needs a hand on the cable.
        PICO_PANIC_FUNCTION=pelagic_panic)
    target_compile_options(pelagic PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(pelagic pico_stdlib pico_multicore pico_scanvideo_dpi hardware_pwm hardware_dma m)
    if(PELAGIC_INTERP)
        target_link_libraries(pelagic hardware_interp)
    endif()
    pico_enable_stdio_usb(pelagic 1)
    # UART defaults collide with VGA. USB CDC is the sole telemetry channel.
    pico_enable_stdio_uart(pelagic 0)
    pico_add_extra_outputs(pelagic)
endif()
