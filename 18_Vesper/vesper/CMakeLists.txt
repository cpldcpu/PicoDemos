cmake_minimum_required(VERSION 3.13)
option(VESPER_HOST "Build the SDL2 desktop player" OFF)
if(VESPER_HOST)
    project(vesper C)
    set(CMAKE_C_STANDARD 11)
    find_package(SDL2 CONFIG REQUIRED)
    add_executable(vesper host/main.c render.c score.c synth.c)
    target_include_directories(vesper PRIVATE .)
    target_compile_options(vesper PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(vesper PRIVATE SDL2::SDL2 m)
    add_executable(vesper_check tools/check.c render.c score.c synth.c)
    target_include_directories(vesper_check PRIVATE .)
    target_compile_options(vesper_check PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation -ftrapv)
    target_link_libraries(vesper_check PRIVATE m)
    enable_testing()
    add_test(NAME whole_production COMMAND vesper_check)
else()
    set(PICO_BOARD pico2 CACHE STRING "RP2350 board")
    set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Cortex-M33 with FPU")
    if(NOT PICO_SDK_PATH)
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    endif()
    if(NOT PICO_EXTRAS_PATH)
        set(PICO_EXTRAS_PATH "$ENV{PICO_EXTRAS_PATH}")
    endif()
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
    include(${PICO_EXTRAS_PATH}/external/pico_extras_import.cmake)
    project(vesper C CXX ASM)
    set(CMAKE_C_STANDARD 11)
    pico_sdk_init()
    if(NOT PICO_PLATFORM STREQUAL "rp2350-arm-s")
        message(FATAL_ERROR "VESPER requires RP2350 Cortex-M33 (rp2350-arm-s)")
    endif()
    add_executable(vesper main.c video.c audio_pwm.c render.c score.c synth.c)
    target_include_directories(vesper PRIVATE .)
    target_compile_definitions(vesper PRIVATE PICO_BUILD=1
        PICO_SCANVIDEO_MAX_SCANLINE_BUFFER_WORDS=324
        PICO_SCANVIDEO_SCANLINE_BUFFER_COUNT=12
        PICO_STACK_SIZE=0x1000 PICO_CORE1_STACK_SIZE=0x1000)
    target_compile_options(vesper PRIVATE -O2 -Wall -Wextra -Wno-misleading-indentation)
    target_link_libraries(vesper pico_stdlib pico_multicore pico_scanvideo_dpi hardware_pwm hardware_dma m)
    pico_enable_stdio_usb(vesper 1)
    # UART defaults collide with VGA. USB CDC is the sole telemetry channel.
    pico_enable_stdio_uart(vesper 0)
    pico_add_extra_outputs(vesper)
endif()
